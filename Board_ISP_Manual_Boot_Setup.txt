dmesg -n 3
OVERRIDE=/mnt/sd/isp_runtime_override
CFG=$OVERRIDE/isp_rtd6122c_0.cfg
BACKUP_DIR=/mnt/sd/isp_original_backup

if grep -q ' /mnt/app/isp ' /proc/self/mountinfo; then
    umount /mnt/app/isp
fi

SOURCE=/mnt/app/isp/isp_rtd6122c_0.cfg

if [ ! -f "$SOURCE" ]; then
    echo "ERROR: original CFG not found: $SOURCE"
else
    mkdir -p "$OVERRIDE" "$BACKUP_DIR"
    cp -a /mnt/app/isp/. "$OVERRIDE"/
    cp -p "$SOURCE" "$BACKUP_DIR/isp_rtd6122c_0_original.cfg"

    awk 'BEGIN{left="fcurve_left_lut=";for(i=0;i<65;i++){v=(i<32?i*2048:65535);left=left (i?",":"") v}right="fcurve_right_lut=";for(i=0;i<17;i++){right=right (i?",":"") 65535}} /^\[/{section=$0} /^fcurve_left_lut=/{print left;next} /^fcurve_right_lut=/{print right;next} section=="[EDGE]"&&/^enable=/{print "enable=0";next} section=="[POST_SHARPEN_1]"&&/^enable=/{print "enable=0";next} section=="[POST_SHARPEN_2]"&&/^enable=/{print "enable=0";next} {print}' "$SOURCE" > "$CFG.new"

    if [ "$(grep -c '^fcurve_left_lut=' "$CFG.new")" -eq 1 ] && [ "$(grep -c '^fcurve_right_lut=' "$CFG.new")" -eq 1 ]; then
        mv "$CFG.new" "$CFG"
        sync
        mount --bind "$OVERRIDE" /mnt/app/isp

        awk -F= '/^fcurve_left_lut=/{n=split($2,a,",");print "Left count="n,"Left[0]="a[1],"Left[31]="a[32],"Left[32]="a[33],"Left[64]="a[65]} /^fcurve_right_lut=/{n=split($2,a,",");print "Right count="n,"Right[0]="a[1],"Right[16]="a[17]}' /mnt/app/isp/isp_rtd6122c_0.cfg

        awk '/^\[/{section=$0} /^enable=/{if(section=="[EDGE]"||section=="[POST_SHARPEN_1]"||section=="[POST_SHARPEN_2]")print section,$0} section=="[NR]"&&/^filter_enable=/{print section,$0}' /mnt/app/isp/isp_rtd6122c_0.cfg

        cmp "$CFG" /mnt/app/isp/isp_rtd6122c_0.cfg >/dev/null && echo "Runtime ISP CFG PASS"
    else
        echo "ERROR: generated CFG validation failed"
        rm -f "$CFG.new"
    fi
fi
